# Documentation toolchain for the published site.
# Install with:  pip install -r requirements.txt
#
# Version ranges are pinned to a major version so the site keeps building
# for years without surprise breakage, while still receiving patch fixes.

# Obsidian vault and the published site are literally the same Markdown.

# Directory-driven navigation: no hand-maintained `nav:` block to update
# every time a note is added. Section titles come from `.pages` files.
mkdocs-awesome-pages-plugin>=2.9,<3.0

# Used by the publishing hook to parse YAML frontmatter.
PyYAML>=6.0,<7.0
